def add(a, b):
    return a + b

def subtract(a, b):
    return a - b

def multiply(a, b):
    return a * b

def divide(a, b):
    if b == 0:
        return "Error! Division by zero."
    return a / b

def km_to_miles(km):
    return km * 0.621371

def c_to_f(celsius):
    return (celsius * 9/5) + 32

def usd_to_inr(usd):
    # Example conversion rate
    return usd * 83.0

def calculator():
    while True:
        print("\n--- Interactive Calculator & Unit Converter ---")
        print("1. Basic Arithmetic (+, -, *, /)")
        print("2. Unit Conversion (Km to Miles, Celsius to Fahrenheit)")
        print("3. Currency Conversion (USD to INR)")
        print("4. Exit")
        
        choice = input("Enter your choice (1-4): ")
        
        if choice == '1':
            print("\nSelect Operation:")
            print("'+' : Addition")
            print("'-' : Subtraction")
            print("'*' : Multiplication")
            print("'/' : Division")
            
            op = input("Enter operator: ")
            if op not in ('+', '-', '*', '/'):
                print("Invalid operator. Please try again.")
                continue
                
            try:
                num1 = float(input("Enter first number: "))
                num2 = float(input("Enter second number: "))
            except ValueError:
                print("Invalid input. Please enter numbers only.")
                continue
                
            if op == '+':
                print(f"Result: {num1} + {num2} = {add(num1, num2)}")
            elif op == '-':
                print(f"Result: {num1} - {num2} = {subtract(num1, num2)}")
            elif op == '*':
                print(f"Result: {num1} * {num2} = {multiply(num1, num2)}")
            elif op == '/':
                print(f"Result: {num1} / {num2} = {divide(num1, num2)}")
                
        elif choice == '2':
            print("\nUnit Conversion:")
            print("1. Kilometers to Miles")
            print("2. Celsius to Fahrenheit")
            sub_choice = input("Enter sub-choice (1-2): ")
            
            try:
                val = float(input("Enter value to convert: "))
            except ValueError:
                print("Invalid input. Please enter a valid number.")
                continue
                
            if sub_choice == '1':
                print(f"Result: {val} km = {km_to_miles(val):.2f} miles")
            elif sub_choice == '2':
                print(f"Result: {val}°C = {c_to_f(val):.2f}°F")
            else:
                print("Invalid sub-choice.")
                
        elif choice == '3':
            try:
                usd = float(input("Enter amount in USD: "))
                print(f"Result: ${usd} = ₹{usd_to_inr(usd):.2f}")
            except ValueError:
                print("Invalid input. Please enter a number.")
                
        elif choice == '4':
            print("Exiting program. Goodbye!")
            break
        else:
            print("Invalid choice. Please select between 1 and 4.")

if __name__ == "__main__":
    calculator()
